% ---------------------------------------------------------------- contents
% A section number in the contents sits in a fixed-width box, and book.cls
% sizes that box for one digit either side of the point. This book has
% thirty-nine chapters and several with more than nine sections, so entries
% like `36.10` overrun the box and collide with the title. Widen it enough for
% the worst case the book can produce (two digits, point, two digits).
% The page-number column is sized for two digits; this book runs past 300, so
% every three-digit entry overran the text block by 1.3pt. Widen the column and
% the right margin that clears it by the same amount.
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3em}}
\renewcommand{\@pnumwidth}{2.2em}
\renewcommand{\@tocrmarg}{3.2em}
\makeatother

% Prose here is full of unbreakable identifiers set in \texttt. Allow the
% typewriter face to hyphenate, so a long identifier at a line end wraps
% instead of running into the margin.
\usepackage[htt]{hyphenat}

% Link colour. meta.yaml cannot carry an RGB triple: pandoc drops whatever
% `linkcolor:` holds straight into \hypersetup, so the YAML list [0,0.25,0.5]
% arrived as the colour NAME `00.250.5` and xcolor refused it. It went unnoticed
% for as long as the book had no links at all — the first `[text](url)` written
% into a chapter failed the build. Named here (xcolor is already loaded by the
% time this header is included), and meta.yaml refers to it by name.
\definecolor{rakuppblue}{rgb}{0,0.25,0.5}
