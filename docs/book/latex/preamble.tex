% Code blocks: pandoc wraps highlighted listings in `Shaded` (from `framed`).
% Tint the panel and set the listing a size down, so it reads as an object on
% the page rather than as indented prose.
\definecolor{shadecolor}{gray}{0.965}
\makeatletter
\AtBeginDocument{%
  \@ifundefined{Shaded}{}{%
    \let\rkpp@Shaded\Shaded
    \let\rkpp@endShaded\endShaded
    \renewenvironment{Shaded}{\small\rkpp@Shaded}{\rkpp@endShaded}}%
}
\makeatother

% Tables are set a size down and may break across pages.
